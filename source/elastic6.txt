import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;

import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.*;

public class ElasticsearchToExcel {

    private static final String EXCEL_DIR = "exceldir";

    public static void main(String[] args) throws Exception {
        // 어제 데이터
        Map<String, Map<String, Long>> yesterdayData = new LinkedHashMap<>();
        Map<String, Long> key1Yesterday = new LinkedHashMap<>();
        key1Yesterday.put("a", 10L);
        key1Yesterday.put("b", 20L);
        key1Yesterday.put("c", 30L);
        yesterdayData.put("key1", key1Yesterday);

        Map<String, Long> key2Yesterday = new LinkedHashMap<>();
        key2Yesterday.put("a", 5L);
        key2Yesterday.put("b", 2L);
        key2Yesterday.put("c", 10L);
        yesterdayData.put("key2", key2Yesterday);

        // 오늘 데이터 (2일전)
        Map<String, Map<String, Long>> todayData = new LinkedHashMap<>();
        Map<String, Long> key1Today = new LinkedHashMap<>();
        key1Today.put("a", 15L);
        key1Today.put("b", 25L);
        key1Today.put("c", 35L);
        todayData.put("key1", key1Today);

        Map<String, Long> key3Today = new LinkedHashMap<>();
        key3Today.put("a", 8L);
        key3Today.put("b", 12L);
        key3Today.put("c", 20L);
        todayData.put("key3", key3Today);

        // 엑셀 생성
        createExcelWithComparison(yesterdayData, todayData, "service_comparison.xlsx");
    }

    /**
     * 디렉토리가 없으면 생성
     */
    private static void ensureDirectoryExists(String dirPath) throws IOException {
        Path path = Paths.get(dirPath);
        if (!Files.exists(path)) {
            Files.createDirectories(path);
            System.out.println("디렉토리 생성: " + path.toAbsolutePath());
        }
    }

    /**
     * 두 날짜의 데이터를 비교하여 엑셀 생성
     * @param yesterdayData 어제 데이터
     * @param twoDaysAgoData 2일전 데이터
     * @param fileName 파일명
     */
    public static void createExcelWithComparison(
            Map<String, Map<String, Long>> yesterdayData,
            Map<String, Map<String, Long>> twoDaysAgoData,
            String fileName) throws Exception {

        // 디렉토리 생성
        ensureDirectoryExists(EXCEL_DIR);

        // 전체 파일 경로
        String filePath = EXCEL_DIR + File.separator + fileName;

        Workbook workbook = new XSSFWorkbook();
        Sheet sheet = workbook.createSheet("서비스 집계");

        // 스타일 설정
        CellStyle headerStyle = createHeaderStyle(workbook);

        // 모든 키 수집 (어제 + 2일전)
        Set<String> allKeys = new LinkedHashSet<>();
        allKeys.addAll(yesterdayData.keySet());
        allKeys.addAll(twoDaysAgoData.keySet());

        // 모든 서비스 수집 (a, b, c)
        Set<String> services = new LinkedHashSet<>();
        yesterdayData.values().forEach(map -> services.addAll(map.keySet()));
        twoDaysAgoData.values().forEach(map -> services.addAll(map.keySet()));
        List<String> serviceList = new ArrayList<>(services);

        // 헤더 행 생성
        Row headerRow = sheet.createRow(0);

        // "key" 셀
        Cell keyCell = headerRow.createCell(0);
        keyCell.setCellValue("key");
        keyCell.setCellStyle(headerStyle);

        int col = 1;

        // "어제" 헤더
        Cell yesterdayCell = headerRow.createCell(col);
        yesterdayCell.setCellValue("어제");
        yesterdayCell.setCellStyle(headerStyle);

        // 어제 서비스별 헤더 (a, b, c)
        for (int i = 0; i < serviceList.size(); i++) {
            Cell cell = headerRow.createCell(col + i);
            cell.setCellValue(serviceList.get(i));
            cell.setCellStyle(headerStyle);
        }
        col += serviceList.size();

        // "2일전" 헤더
        Cell twoDaysAgoCell = headerRow.createCell(col);
        twoDaysAgoCell.setCellValue("2일전");
        twoDaysAgoCell.setCellStyle(headerStyle);

        // 2일전 서비스별 헤더 (a, b, c)
        for (int i = 0; i < serviceList.size(); i++) {
            Cell cell = headerRow.createCell(col + i);
            cell.setCellValue(serviceList.get(i));
            cell.setCellStyle(headerStyle);
        }

        // 데이터 행 생성
        int rowNum = 1;
        for (String key : allKeys) {
            Row row = sheet.createRow(rowNum++);

            // key 컬럼
            row.createCell(0).setCellValue(key);

            col = 1;

            // 어제 데이터
            Map<String, Long> yesterdayServiceData = yesterdayData.getOrDefault(key, Collections.emptyMap());
            for (String service : serviceList) {
                Long count = yesterdayServiceData.getOrDefault(service, 0L);
                Cell cell = row.createCell(col++);
                cell.setCellValue(count);
            }

            // 2일전 데이터
            Map<String, Long> twoDaysAgoServiceData = twoDaysAgoData.getOrDefault(key, Collections.emptyMap());
            for (String service : serviceList) {
                Long count = twoDaysAgoServiceData.getOrDefault(service, 0L);
                Cell cell = row.createCell(col++);
                cell.setCellValue(count);
            }
        }

        // 컬럼 너비 자동 조정
        for (int i = 0; i < headerRow.getLastCellNum(); i++) {
            sheet.autoSizeColumn(i);
        }

        // 파일 저장
        try (FileOutputStream fileOut = new FileOutputStream(filePath)) {
            workbook.write(fileOut);
        }

        workbook.close();
        System.out.println("Excel 파일이 생성되었습니다: " + new File(filePath).getAbsolutePath());
    }

    /**
     * 헤더 스타일 생성
     */
    private static CellStyle createHeaderStyle(Workbook workbook) {
        CellStyle style = workbook.createCellStyle();
        Font font = workbook.createFont();
        font.setBold(true);
        style.setFont(font);
        style.setAlignment(HorizontalAlignment.CENTER);
        style.setFillForegroundColor(IndexedColors.GREY_25_PERCENT.getIndex());
        style.setFillPattern(FillPatternType.SOLID_FOREGROUND);

        // 테두리 추가
        style.setBorderTop(BorderStyle.THIN);
        style.setBorderBottom(BorderStyle.THIN);
        style.setBorderLeft(BorderStyle.THIN);
        style.setBorderRight(BorderStyle.THIN);

        return style;
    }
}
