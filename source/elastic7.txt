import java.util.*;
import java.util.stream.Collectors;

public class DataMergeExample {

    // 데이터를 담을 클래스
    static class RowData {
        String key;           // 1번키, 2번키, 3번키
        String company;       // a회사, b회사, c회사
        Long a;
        Long b;
        Long c;

        public RowData(String key, String company) {
            this.key = key;
            this.company = company;
            this.a = 0L;
            this.b = 0L;
            this.c = 0L;
        }

        @Override
        public String toString() {
            return String.format("%s\t%s\ta:%d\tb:%d\tc:%d",
                key, company, a, b, c);
        }
    }

    // 원본 데이터 형식
    static class SourceData {
        String key;
        Map<String, Object> data;

        public SourceData(String key, Map<String, Object> data) {
            this.key = key;
            this.data = data;
        }
    }

    public static void main(String[] args) {
        // xData 원본
        List<SourceData> xData = Arrays.asList(
            new SourceData("1번키", Map.of("a", 10L, "b", 10L, "c", 10L, "com", "a회사")),
            new SourceData("2번키", Map.of("a", 10L, "b", 10L, "c", 10L, "com", "b회사"))
        );

        // yData 원본
        List<SourceData> yData = Arrays.asList(
            new SourceData("2번키", Map.of("b", 20L, "c", 10L, "com", "b회사")),
            new SourceData("3번키", Map.of("b", 10L, "com", "c회사"))
        );

        // zData 원본
        List<SourceData> zData = Arrays.asList(
            new SourceData("3번키", Map.of("c", 10L, "com", "c회사"))
        );

        // 데이터 병합 및 변환
        Map<String, RowData> mergedMap = new LinkedHashMap<>();

        // xData 처리
        processData(xData, mergedMap, "x");

        // yData 처리
        processData(yData, mergedMap, "y");

        // zData 처리
        processData(zData, mergedMap, "z");

        // 최종 결과 리스트
        List<RowData> xDataList = new ArrayList<>();
        List<RowData> yDataList = new ArrayList<>();
        List<RowData> zDataList = new ArrayList<>();

        // 각 데이터셋별로 분리
        for (RowData row : mergedMap.values()) {
            RowData xRow = new RowData(row.key, row.company);
            RowData yRow = new RowData(row.key, row.company);
            RowData zRow = new RowData(row.key, row.company);

            // 원본 데이터에서 값 채우기
            fillFromSource(xData, xRow);
            fillFromSource(yData, yRow);
            fillFromSource(zData, zRow);

            xDataList.add(xRow);
            yDataList.add(yRow);
            zDataList.add(zRow);
        }

        // 결과 출력
        System.out.println("=== xData ===");
        xDataList.forEach(System.out::println);

        System.out.println("\n=== yData ===");
        yDataList.forEach(System.out::println);

        System.out.println("\n=== zData ===");
        zDataList.forEach(System.out::println);

        // 엑셀 형식으로 출력
        System.out.println("\n=== Excel 형식 ===");
        printExcelFormat(xDataList, yDataList, zDataList);
    }

    private static void processData(List<SourceData> sourceList,
                                   Map<String, RowData> mergedMap,
                                   String dataType) {
        for (SourceData source : sourceList) {
            if (!mergedMap.containsKey(source.key)) {
                String company = (String) source.data.get("com");
                mergedMap.put(source.key, new RowData(source.key, company));
            }
        }
    }

    private static void fillFromSource(List<SourceData> sourceList, RowData row) {
        for (SourceData source : sourceList) {
            if (source.key.equals(row.key)) {
                row.a = (Long) source.data.getOrDefault("a", 0L);
                row.b = (Long) source.data.getOrDefault("b", 0L);
                row.c = (Long) source.data.getOrDefault("c", 0L);
                break;
            }
        }
    }

    private static void printExcelFormat(List<RowData> xData,
                                        List<RowData> yData,
                                        List<RowData> zData) {
        System.out.println("\t\txData\t\t\tyData\t\t\tzData");
        System.out.println("\t회사\ta\tb\tc\ta\tb\tc\ta\tb\tc");

        Set<String> allKeys = new LinkedHashSet<>();
        xData.forEach(r -> allKeys.add(r.key));
        yData.forEach(r -> allKeys.add(r.key));
        zData.forEach(r -> allKeys.add(r.key));

        for (String key : allKeys) {
            RowData x = findByKey(xData, key);
            RowData y = findByKey(yData, key);
            RowData z = findByKey(zData, key);

            System.out.printf("%s\t%s\t%d\t%d\t%d\t%d\t%d\t%d\t%d\t%d\t%d\n",
                key,
                x != null ? x.company : "",
                x != null ? x.a : 0,
                x != null ? x.b : 0,
                x != null ? x.c : 0,
                y != null ? y.a : 0,
                y != null ? y.b : 0,
                y != null ? y.c : 0,
                z != null ? z.a : 0,
                z != null ? z.b : 0,
                z != null ? z.c : 0
            );
        }
    }

    private static RowData findByKey(List<RowData> list, String key) {
        return list.stream()
                   .filter(r -> r.key.equals(key))
                   .findFirst()
                   .orElse(null);
    }
}
