import java.util.*;

public class DataMergeExample {

    // 데이터를 담을 클래스
    static class RowData {
        String key;           // 1번키, 2번키, 3번키
        String company;       // a회사, b회사, c회사
        String serviceName;   // a회사서비스, b회사서비스, c회사서비스
        Long a;
        Long b;
        Long c;

        public RowData(String key, String company, String serviceName, Long a, Long b, Long c) {
            this.key = key;
            this.company = company;
            this.serviceName = serviceName;
            this.a = a != null ? a : 0L;
            this.b = b != null ? b : 0L;
            this.c = c != null ? c : 0L;
        }

        // 복사 생성자 (값만 0으로 초기화)
        public RowData(RowData other) {
            this.key = other.key;
            this.company = other.company;
            this.serviceName = other.serviceName;
            this.a = 0L;
            this.b = 0L;
            this.c = 0L;
        }

        @Override
        public String toString() {
            return String.format("%s\t%s\t%s\ta:%d\tb:%d\tc:%d",
                key, company, serviceName, a, b, c);
        }
    }

    public static void main(String[] args) {
        // 원본 데이터 - 이미 List<RowData> 형식
        List<RowData> xData = Arrays.asList(
            new RowData("1번키", "a회사", "a회사서비스", 10L, 10L, 10L),
            new RowData("2번키", "b회사", "b회사서비스", 10L, 10L, 10L)
        );

        List<RowData> yData = Arrays.asList(
            new RowData("2번키", "b회사", "b회사서비스", null, 20L, 10L),
            new RowData("3번키", "c회사", "c회사서비스", null, 10L, null)
        );

        List<RowData> zData = Arrays.asList(
            new RowData("3번키", "c회사", "c회사서비스", null, null, 10L)
        );

        // 모든 키 수집 (순서 유지)
        Map<String, RowData> allKeysMap = new LinkedHashMap<>();
        collectKeys(xData, allKeysMap);
        collectKeys(yData, allKeysMap);
        collectKeys(zData, allKeysMap);

        // 각 데이터셋을 전체 키 기준으로 재구성
        List<RowData> xDataFull = fillMissingRows(allKeysMap, xData);
        List<RowData> yDataFull = fillMissingRows(allKeysMap, yData);
        List<RowData> zDataFull = fillMissingRows(allKeysMap, zData);

        // 결과 출력
        System.out.println("=== xData ===");
        xDataFull.forEach(System.out::println);

        System.out.println("\n=== yData ===");
        yDataFull.forEach(System.out::println);

        System.out.println("\n=== zData ===");
        zDataFull.forEach(System.out::println);

        // 엑셀 형식으로 출력
        System.out.println("\n=== Excel 형식 ===");
        printExcelFormat(xDataFull, yDataFull, zDataFull);
    }

    // 모든 키의 기본 정보 수집 (key, company, serviceName)
    private static void collectKeys(List<RowData> dataList, Map<String, RowData> allKeysMap) {
        for (RowData row : dataList) {
            if (!allKeysMap.containsKey(row.key)) {
                allKeysMap.put(row.key, row);
            }
        }
    }

    // 누락된 키는 0으로 채워서 전체 리스트 생성
    private static List<RowData> fillMissingRows(Map<String, RowData> allKeysMap,
                                                  List<RowData> sourceData) {
        List<RowData> result = new ArrayList<>();

        for (String key : allKeysMap.keySet()) {
            RowData found = sourceData.stream()
                                      .filter(r -> r.key.equals(key))
                                      .findFirst()
                                      .orElse(null);

            if (found != null) {
                result.add(found);
            } else {
                // 해당 키가 없으면 기본 정보만 가져오고 값은 0으로
                RowData baseInfo = allKeysMap.get(key);
                result.add(new RowData(baseInfo.key, baseInfo.company, baseInfo.serviceName,
                                      0L, 0L, 0L));
            }
        }

        return result;
    }

    private static void printExcelFormat(List<RowData> xData,
                                        List<RowData> yData,
                                        List<RowData> zData) {
        System.out.println("\t\t\txData\t\t\t\tyData\t\t\t\tzData");
        System.out.println("\t회사\t서비스명\ta\tb\tc\ta\tb\tc\ta\tb\tc");

        for (int i = 0; i < xData.size(); i++) {
            RowData x = xData.get(i);
            RowData y = yData.get(i);
            RowData z = zData.get(i);

            System.out.printf("%s\t%s\t%s\t%d\t%d\t%d\t%d\t%d\t%d\t%d\t%d\t%d\n",
                x.key,
                x.company,
                x.serviceName,
                x.a, x.b, x.c,
                y.a, y.b, y.c,
                z.a, z.b, z.c
            );
        }
    }

    private static RowData findByKey(List<RowData> list, String key) {
        return list.stream()
                   .filter(r -> r.key.equals(key))
                   .findFirst()
                   .orElse(null);  // 이 경우는 발생하지 않아야 함 (fillMissingRows로 모든 키가 있음)
    }

    private static void cleanCompanyNames(List<RowData> dataList) {
        for (RowData row : dataList) {
            if (row.company != null) {
                row.company = row.company.replaceAll("[\\[\\]\"]", "").trim();
            }
        }
    }


}
