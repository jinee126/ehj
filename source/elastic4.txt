public SearchRequest buildSearchRequestConcise() {
    String scriptSource =
        "String keyword = doc['log.message.service.keyword'].value; " +
        "if (keyword.startsWith('MAP')) return 'MAP'; " +
        "if (keyword.startsWith('PLACE')) return 'PLACE'; " +
        "if (keyword.startsWith('ISOCHRONE')) return 'ISOCHRONE'; " +
        "if (keyword.startsWith('MATRIX')) return 'MATRIX'; " +
        "if (keyword.startsWith('GEOCODING')) return 'GEOCODING'; " +
        "if (keyword.startsWith('PEDESTRIAN') || keyword.startsWith('OPTIMIZATION') || " +
        "   keyword.startsWith('ROUTE') || keyword.startsWith('TRANSIT') || " +
        "   keyword.startsWith('MULTISTOP') || keyword.startsWith('MULTISTOP_2000') || " +
        "   keyword.startsWith('VIRTUALIZATION')) return 'DIRECTION'; " +
        "return 'ETC';";

    return SearchRequest.of(s -> s
        .index("your-index-name")
        .size(0)
        .query(q -> q.bool(b -> b
            .filter(f -> f.term(t -> t.field("log.file.type").value("rest")))
            .filter(f -> f.term(t -> t.field("log.message.status").value("REQUEST")))
        ))
        .aggregations("aggs", a -> a
            .aggregations("KEY_COUNT", ka -> ka
                .terms(t -> t.field("log.message.key").size(1000))
                .aggregations("service_group", sa -> sa
                    .terms(ts -> ts
                        .script(sc -> sc.inline(i -> i
                            .source(scriptSource)
                            .lang("painless")
                        ))
                        .size(10000000)
                    )
                )
            )
        )
    );
}
