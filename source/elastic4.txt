public SearchRequest buildSearchRequestChaining() {
    String scriptSource =
        "String keyword = doc['log.message.service.keyword'].value; " +
        "if (keyword.startsWith('MAP')) return 'MAP'; " +
        "if (keyword.startsWith('PLACE')) return 'PLACE'; " +
        "if (keyword.startsWith('ISOCHRONE')) return 'ISOCHRONE'; " +
        "if (keyword.startsWith('MATRIX')) return 'MATRIX'; " +
        "if (keyword.startsWith('GEOCODING')) return 'GEOCODING'; " +
        "if (keyword.startsWith('PEDESTRIAN') || keyword.startsWith('OPTIMIZATION') || " +
        "   keyword.startsWith('ROUTE') || keyword.startsWith('TRANSIT') || " +
        "   keyword.startsWith('MULTISTOP') || keyword.startsWith('MULTISTOP_2000') || " +
        "   keyword.startsWith('VIRTUALIZATION')) return 'DIRECTION'; " +
        "return 'ETC';";

    return new SearchRequest.Builder()
        .index("your-index-name")
        .size(0)
        .query(new Query.Builder()
            .bool(new BoolQuery.Builder()
                .filter(new Query.Builder()
                    .term(new TermQuery.Builder()
                        .field("log.file.type")
                        .value("rest")
                        .build())
                    .build())
                .filter(new Query.Builder()
                    .term(new TermQuery.Builder()
                        .field("log.message.status")
                        .value("REQUEST")
                        .build())
                    .build())
                .build())
            .build())
        .aggregations("aggs", new Aggregation.Builder()
            .aggregations("KEY_COUNT", new Aggregation.Builder()
                .terms(new TermsAggregation.Builder()
                    .field("log.message.key")
                    .size(1000)
                    .build())
                .aggregations("service_group", new Aggregation.Builder()
                    .terms(new TermsAggregation.Builder()
                        .script(s -> s
                            .inline(i -> i
                                .source(scriptSource)
                                .lang("painless")
                            )
                        )
                        .size(10000000)
                        .build())
                    .build())
                .build())
            .build())
        .build();
}
