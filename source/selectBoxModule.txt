// store/commonCodeAtoms.ts - Jotai atom 정의
import { atom } from 'jotai';
import { atomWithQuery } from 'jotai-tanstack-query';

export interface CommonCode {
  code: string;
  name: string;
}

// 방법 1: atomWithQuery 사용 (추천 - 자동 캐싱)
export const categoryCodesAtom = atomWithQuery(() => ({
  queryKey: ['commonCode', 'category'],
  queryFn: async () => {
    const res = await fetch('/api/common-code/category');
    return res.json() as Promise<CommonCode[]>;
  },
}));

export const statusCodesAtom = atomWithQuery(() => ({
  queryKey: ['commonCode', 'status'],
  queryFn: async () => {
    const res = await fetch('/api/common-code/status');
    return res.json() as Promise<CommonCode[]>;
  },
}));

// 방법 2: 일반 atom + async atom (더 심플)
export const categoryCodesSimpleAtom = atom<CommonCode[]>([]);
export const statusCodesSimpleAtom = atom<CommonCode[]>([]);

// 공통코드 로드하는 async atom
export const loadCommonCodesAtom = atom(
  null,
  async (get, set) => {
    try {
      const [categoryRes, statusRes] = await Promise.all([
        fetch('/api/common-code/category'),
        fetch('/api/common-code/status')
      ]);

      const [categoryData, statusData] = await Promise.all([
        categoryRes.json(),
        statusRes.json()
      ]);

      set(categoryCodesSimpleAtom, categoryData);
      set(statusCodesSimpleAtom, statusData);
    } catch (error) {
      console.error('공통코드 로드 실패:', error);
    }
  }
);

// 방법 3: 하나의 atom으로 모든 공통코드 관리
export const allCommonCodesAtom = atom<Record<string, CommonCode[]>>({});

export const loadAllCommonCodesAtom = atom(
  null,
  async (get, set, types: string[]) => {
    try {
      const promises = types.map(type =>
        fetch(`/api/common-code/${type}`).then(res => res.json())
      );

      const results = await Promise.all(promises);

      const data = types.reduce((acc, type, index) => {
        acc[type] = results[index];
        return acc;
      }, {} as Record<string, CommonCode[]>);

      set(allCommonCodesAtom, data);
    } catch (error) {
      console.error('공통코드 로드 실패:', error);
    }
  }
);

import { CommonCode } from '@/store/commonCodeAtoms';

interface CommonCodeSelectProps {
  label: string;
  value: string;
  onChange: (value: string) => void;
  options: CommonCode[];
  placeholder?: string;
  required?: boolean;
}

export function CommonCodeSelect({
  label,
  value,
  onChange,
  options,
  placeholder = '선택하세요',
  required = false
}: CommonCodeSelectProps) {
  return (
    <div>
      <label className="block text-sm font-medium mb-2">
        {label}
        {required && <span className="text-red-500 ml-1">*</span>}
      </label>
      <select
        value={value}
        onChange={(e) => onChange(e.target.value)}
        className="w-full px-3 py-2 border border-gray-300 rounded-md"
        required={required}
      >
        <option value="">{placeholder}</option>
        {options.map((item) => (
          <option key={item.code} value={item.code}>
            {item.name}
          </option>
        ))}
      </select>
    </div>
  );
}

// 사용 예시 1: atomWithQuery 사용 (추천)
// ============================================

'use client';

import { useAtom, useSetAtom } from 'jotai';
import {
  categoryCodesAtom,
  statusCodesAtom
} from '@/store/commonCodeAtoms';
import { CommonCodeSelect } from '@/components/CommonCodeSelect';

export default function DetailPage1() {
  const [formData, setFormData] = useState({
    categoryCode: '',
    statusCode: ''
  });

  // Jotai로 공통코드 가져오기
  const [categoryResult] = useAtom(categoryCodesAtom);
  const [statusResult] = useAtom(statusCodesAtom);

  // 로딩 체크
  if (categoryResult.isLoading || statusResult.isLoading) {
    return <div>로딩 중...</div>;
  }

  const categoryList = categoryResult.data || [];
  const statusList = statusResult.data || [];

  return (
    <div className="max-w-2xl mx-auto p-8">
      <form className="space-y-4">
        <CommonCodeSelect
          label="카테고리"
          value={formData.categoryCode}
          onChange={(value) => setFormData({...formData, categoryCode: value})}
          options={categoryList}
        />

        <CommonCodeSelect
          label="상태"
          value={formData.statusCode}
          onChange={(value) => setFormData({...formData, statusCode: value})}
          options={statusList}
        />
      </form>
    </div>
  );
}

'use client';

import { useAtom, useSetAtom, useAtomValue } from 'jotai';
import {
  categoryCodesSimpleAtom,
  statusCodesSimpleAtom,
  loadCommonCodesAtom
} from '@/store/commonCodeAtoms';

export default function DetailPage2() {
  const [formData, setFormData] = useState({
    categoryCode: '',
    statusCode: ''
  });

  // 공통코드 데이터
  const categoryList = useAtomValue(categoryCodesSimpleAtom);
  const statusList = useAtomValue(statusCodesSimpleAtom);

    loadCommonCodes(); // 한 번만 호출

          options={statusList}
        />
      </form>
    </div>
  );
}

'use client';

import { useAtomValue, useSetAtom } from 'jotai';
import {
  allCommonCodesAtom,
  loadAllCommonCodesAtom
} from '@/store/commonCodeAtoms';

export default function DetailPage3() {
  const [formData, setFormData] = useState({
    categoryCode: '',
    statusCode: ''
  });

  // 모든 공통코드
  const commonCodes = useAtomValue(allCommonCodesAtom);
  const loadCodes = useSetAtom(loadAllCommonCodesAtom);

  useEffect(() => {
    loadCodes(['category', 'status']); // 필요한 것만 로드
  }, []);

  return (
    <div className="max-w-2xl mx-auto p-8">
      <form className="space-y-4">
        <CommonCodeSelect
          label="카테고리"
          value={formData.categoryCode}
          onChange={(value) => setFormData({...formData, categoryCode: value})}
          options={commonCodes.category || []}
        />

        <CommonCodeSelect
          label="상태"
          value={formData.statusCode}
          onChange={(value) => setFormData({...formData, statusCode: value})}
          options={commonCodes.status || []}
        />
      </form>
    </div>
  );
}
