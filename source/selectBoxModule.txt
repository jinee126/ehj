// store/commonCodeAtoms.ts
import { atom } from 'jotai';

export interface CommonCode {
  code: string;
  name: string;
}

// 모든 공통코드를 저장하는 atom
export const allCommonCodesAtom = atom<Record<string, CommonCode[]>>({});

// 공통코드 로드하는 atom (파라미터를 받음!)
export const loadCommonCodesAtom = atom(
  null,  // 읽기 값 없음
  async (get, set, types: string[]) => {  // 👈 types 파라미터로 받음!
    try {
      // 이미 로드된 코드는 제외 (중복 호출 방지)
      const currentCodes = get(allCommonCodesAtom);
      const needToLoad = types.filter(type => !currentCodes[type]);

      if (needToLoad.length === 0) {
        console.log('이미 모든 코드가 로드되어 있음');
        return; // 이미 다 있으면 API 호출 안 함
      }

      console.log('로드할 공통코드:', needToLoad);

      // 필요한 것만 API 호출
      const promises = needToLoad.map(type =>
        fetch(`/api/common-code/${type}`).then(res => res.json())
      );

      const results = await Promise.all(promises);

      // 새로운 데이터 추가 (기존 데이터 유지)
      const newData = needToLoad.reduce((acc, type, index) => {
        acc[type] = results[index];
        return acc;
      }, {} as Record<string, CommonCode[]>);

      set(allCommonCodesAtom, {
        ...currentCodes,  // 기존 데이터 유지
        ...newData        // 새 데이터 추가
      });
    } catch (error) {
      console.error('공통코드 로드 실패:', error);
    }
  }
);

// ============================================
// 사용 예시 1: 페이지 A - menu, auth 필요
// ============================================

'use client';

import { useEffect, useState } from 'react';
import { useAtomValue, useSetAtom } from 'jotai';
import { allCommonCodesAtom, loadCommonCodesAtom } from '@/store/commonCodeAtoms';

export default function PageA() {
  const [formData, setFormData] = useState({
    menuCode: '',
    authCode: ''
  });

  // 1. 공통코드 데이터 가져오기
  const commonCodes = useAtomValue(allCommonCodesAtom);

  // 2. 로드 함수 가져오기
  const loadCodes = useSetAtom(loadCommonCodesAtom);

  useEffect(() => {
    // 3. 이 페이지에 필요한 공통코드만 로드
    loadCodes(['menu', 'auth']);  // 👈 파라미터로 넘김!
  }, []);

  return (
    <div className="p-8">
      <h1>페이지 A (menu, auth 사용)</h1>

      <div className="space-y-4">
        <div>
          <label>메뉴</label>
          <select
            value={formData.menuCode}
            onChange={(e) => setFormData({...formData, menuCode: e.target.value})}
            className="w-full px-3 py-2 border rounded"
          >
            <option value="">선택하세요</option>
            {commonCodes.menu?.map(item => (
              <option key={item.code} value={item.code}>
                {item.name}
              </option>
            ))}
          </select>
        </div>

        <div>
          <label>권한</label>
          <select
            value={formData.authCode}
            onChange={(e) => setFormData({...formData, authCode: e.target.value})}
            className="w-full px-3 py-2 border rounded"
          >
            <option value="">선택하세요</option>
            {commonCodes.auth?.map(item => (
              <option key={item.code} value={item.code}>
                {item.name}
              </option>
            ))}
          </select>
        </div>
      </div>
    </div>
  );
}

/*
1. 페이지 A 진입:
   loadCodes(['menu', 'auth']) 실행
   → API 호출: /api/common-code/menu, /api/common-code/auth
   → allCommonCodesAtom에 저장:
     {
       menu: [...],
       auth: [...]
     }

2. 페이지 B 진입:
   loadCodes(['category', 'status']) 실행
   → API 호출: /api/common-code/category, /api/common-code/status
   → allCommonCodesAtom에 추가 저장:
     {
       menu: [...],        // 기존 데이터 유지
       auth: [...],        // 기존 데이터 유지
       category: [...],    // 새로 추가
       status: [...]       // 새로 추가
     }

3. 페이지 A로 다시 돌아감:
   loadCodes(['menu', 'auth']) 실행
   → 이미 로드되어 있음!
   → API 호출 안 함 (캐싱됨)
   → 바로 사용

4. 페이지 C 진입:
   loadCodes(['menu']) 실행
   → menu는 이미 있음!
   → API 호출 안 함
*/

// ============================================
// Custom Hook으로 만들기 (선택사항)
// ============================================

import { useEffect } from 'react';
import { useAtomValue, useSetAtom } from 'jotai';
import { allCommonCodesAtom, loadCommonCodesAtom } from '@/store/commonCodeAtoms';

export function useCommonCodes(types: string[]) {
  const commonCodes = useAtomValue(allCommonCodesAtom);
  const loadCodes = useSetAtom(loadCommonCodesAtom);

  useEffect(() => {
    loadCodes(types);
  }, [types.join(',')]);

  // 필요한 코드만 반환
  const codes = types.reduce((acc, type) => {
    acc[type] = commonCodes[type] || [];
    return acc;
  }, {} as Record<string, CommonCode[]>);

  return codes;
}

// 사용 (더 간단!)
export default function PageD() {
  const codes = useCommonCodes(['menu', 'auth']);

  return (
    <select>
      {codes.menu?.map(item => (
        <option key={item.code}>{item.name}</option>
      ))}
    </select>
  );
}
