// store/commonCodeAtoms.ts
import { atom } from 'jotai';
import { atomFamily } from 'jotai/utils';

export interface CommonCode {
  code: string;
  name: string;
}

// ============================================
// 방법 1: atomFamily 사용 (추천 ⭐)
// ============================================

// 타입별로 자동으로 개별 atom 생성
export const commonCodeFamily = atomFamily((type: string) =>
  atom<CommonCode[]>([])
);

// 로드 함수도 타입별로 생성
export const loadCommonCodeFamily = atomFamily((type: string) =>
  atom(
    null,
    async (get, set) => {
      // 해당 타입의 atom 가져오기
      const codeAtom = commonCodeFamily(type);
      const current = get(codeAtom);

      // 이미 있으면 API 호출 안 함
      if (current.length > 0) {
        console.log(`✅ ${type} 이미 로드됨`);
        return;
      }

      console.log(`📡 ${type} API 호출`);
      const response = await fetch(`/api/common-code/${type}`);
      const data = await response.json();
      set(codeAtom, data);
    }
  )
);

// 사용 예시 1: 기본 사용법
// ============================================

'use client';

import { useAtomValue, useSetAtom } from 'jotai';
import { commonCodeFamily, loadCommonCodeFamily } from '@/store/commonCodeAtoms';

export default function PageA() {
  const [formData, setFormData] = useState({
    menuCode: '',
    authCode: ''
  });

  // 파라미터로 원하는 공통코드 가져오기
  const menuCodes = useAtomValue(commonCodeFamily('menu'));
  const authCodes = useAtomValue(commonCodeFamily('auth'));

  // 로드 함수도 파라미터로
  const loadMenu = useSetAtom(loadCommonCodeFamily('menu'));
  const loadAuth = useSetAtom(loadCommonCodeFamily('auth'));

  useEffect(() => {
    loadMenu();  // menu 로드
    loadAuth();  // auth 로드
  }, []);

  return (
    <div className="p-8 space-y-4">
      <h1>페이지 A (menu, auth)</h1>

      <div>
        <label>메뉴</label>
        <select
          value={formData.menuCode}
          onChange={(e) => setFormData({...formData, menuCode: e.target.value})}
          className="w-full px-3 py-2 border rounded"
        >
          <option value="">선택하세요</option>
          {menuCodes.map(item => (
            <option key={item.code} value={item.code}>
              {item.name}
            </option>
          ))}
        </select>
      </div>

      <div>
        <label>권한</label>
        <select
          value={formData.authCode}
          onChange={(e) => setFormData({...formData, authCode: e.target.value})}
          className="w-full px-3 py-2 border rounded"
        >
          <option value="">선택하세요</option>
          {authCodes.map(item => (
            <option key={item.code} value={item.code}>
              {item.name}
            </option>
          ))}
        </select>
      </div>
    </div>
  );
}

// Custom Hook으로 만들기 (더 편리하게!)
// ============================================

import { useEffect } from 'react';
import { useAtomValue, useSetAtom } from 'jotai';
import { commonCodeFamily, loadCommonCodeFamily } from '@/store/commonCodeAtoms';

// 파라미터를 받는 Custom Hook
export function useCommonCode(type: string) {
  const codes = useAtomValue(commonCodeFamily(type));
  const loadCodes = useSetAtom(loadCommonCodeFamily(type));

  useEffect(() => {
    loadCodes();
  }, [type]);

  return codes;
}

// 여러 개를 한번에
export function useCommonCodes(types: string[]) {
  const codes: Record<string, CommonCode[]> = {};

  types.forEach(type => {
    // eslint-disable-next-line react-hooks/rules-of-hooks
    codes[type] = useCommonCode(type);
  });

  return codes;
}

/*
atomFamily 동작 방식:

1. commonCodeFamily('menu') 호출
   → 'menu'라는 키로 atom 생성 (처음이면)
   → 이미 있으면 기존 atom 반환

2. commonCodeFamily('auth') 호출
   → 'auth'라는 키로 새로운 atom 생성
   → menu atom과는 완전히 독립적

3. 각 atom은 독립적으로 관리됨
   {
     'menu': menuAtom (독립적인 상태),
     'auth': authAtom (독립적인 상태),
     'category': categoryAtom (독립적인 상태)
   }

4. menu가 업데이트되어도 auth를 사용하는 컴포넌트는 리렌더링 안 됨!
*/
